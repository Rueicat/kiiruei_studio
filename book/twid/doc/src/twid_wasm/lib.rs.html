36// 主要邏輯
37fn validate_inner(id: &str) -> Result<(), &'static str> {
38    // 確認長度 = 10
39    if id.len() != 10 {
40        return Err("輸入長度需要10個字元");
41    }
42
43    let mut chars = id.chars();    // 取unicode
44    
45    //取字母轉數字(要大寫)
46    let first_letter = chars.next().ok_or("empty")?;
47    if !first_letter.is_ascii_uppercase() {
48        return Err("請輸入大寫字母A-Z在開頭");
49    }
50    let code_num = code(first_letter);
51    if code_num == 0 {          //理論不會出現, 但如果輸入36個字母以外的字母情況?
52        return Err("你輸入的字母錯誤");
53    }
54
55    // 各個位數的權重
56    let weight: [u32; 11] = [1,9,8,7,6,5,4,3,2,1,1];  //長度11, 全部都u32的array
57
58    //依照判斷邏輯拆解字母
59    let tens = (code_num / 10) as u32;
60    let ones = (code_num % 10) as u32;
61
62    //依照規定的邏輯處理字母權重
63    let mut sum: u32 = tens * weight[0] + ones * weight[1];
64
65    //後面9個數字, 性別判斷
66    for (i, ch) in chars.enumerate() {
67        
68        let d = ch.to_digit(10).ok_or("字母後面要輸入數字")?;
69        if i == 0 && d != 1 && d != 2 {
70            return Err("性別識別錯誤, 開頭數字需要為1或2");
71        }
72        sum += d * weight[i + 2];
73    }
74    if sum % 10 == 0 {
75        Ok(())
76    } else {
77        Err("身份證號碼驗證失敗")
78    }
79}
